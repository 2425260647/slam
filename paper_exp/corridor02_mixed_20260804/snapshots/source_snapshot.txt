+void UpdateLidarOnlyObservabilityMetric(
+    ceres::Problem* const problem,
+    const ceres::ResidualBlockId occupied_space_residual_block,
+    DirectionalDegeneracyMetric* const metric) {
+  metric->lidar_only_observability_enabled = true;
+  if (occupied_space_residual_block == nullptr) {
+    return;
+  }
+
+  ceres::Problem::EvaluateOptions evaluate_options;
+  evaluate_options.residual_blocks.push_back(occupied_space_residual_block);
+  evaluate_options.apply_loss_function = false;
+  ceres::CRSMatrix jacobian;
+  if (!problem->Evaluate(evaluate_options, nullptr, nullptr, nullptr,
+                         &jacobian) ||
+      jacobian.num_cols != 3 || jacobian.num_rows <= 0 ||
+      jacobian.rows.size() != static_cast<size_t>(jacobian.num_rows + 1)) {
+    return;
+  }
+
+  Eigen::Matrix3d information = Eigen::Matrix3d::Zero();
+  for (int row = 0; row < jacobian.num_rows; ++row) {
+    Eigen::Vector3d jacobian_row = Eigen::Vector3d::Zero();
+    for (int index = jacobian.rows[row]; index < jacobian.rows[row + 1];
+         ++index) {
+      const int column = jacobian.cols[index];
+      if (column >= 0 && column < 3 && std::isfinite(jacobian.values[index])) {
+        jacobian_row[column] = jacobian.values[index];
+      }
+    }
+    information.noalias() += jacobian_row * jacobian_row.transpose();
+  }
+
+  const Eigen::Matrix2d translation_information =
+      0.5 * (information.topLeftCorner<2, 2>() +
+             information.topLeftCorner<2, 2>().transpose());
+  Eigen::SelfAdjointEigenSolver<Eigen::Matrix2d> solver(
+      translation_information);
+  if (solver.info() != Eigen::Success || !solver.eigenvalues().allFinite() ||
+      !solver.eigenvectors().allFinite()) {
+    return;
+  }
+
+  const double minimum_information = std::max(0., solver.eigenvalues()(0));
+  const double maximum_information = std::max(0., solver.eigenvalues()(1));
+  constexpr double kInformationEpsilon = 1e-12;
+  if (maximum_information <= kInformationEpsilon) {
+    return;
+  }
+  metric->lidar_only_observability_valid = true;
+  metric->lidar_only_translation_min_information = minimum_information;
+  metric->lidar_only_translation_max_information = maximum_information;
+  metric->lidar_only_translation_condition_number =
+      maximum_information / std::max(kInformationEpsilon, minimum_information);
+  metric->lidar_only_yaw_information = std::max(0., information(2, 2));
+  metric->lidar_only_weak_translation_direction =
+      solver.eigenvectors().col(0).normalized();
+}

+## 2026-08-03 同类型 2D LiDAR SLAM 联网检索
+
+- 按“2D LiDAR + 轮式里程计、无 IMU”筛选了公开论文、ROS Wiki、ROS Index 和官方 GitHub 仓库。2022 年 Sensors 对比研究（DOI `10.3390/s22186903`）明确将 Cartographer、GMapping、Hector-SLAM、Karto-SLAM 和 RTAB-Map 放在同一 2D SLAM 对比框架中，但这不代表它们的传感器输入完全相同。
+- GMapping 的原始论文为 Grisetti、Stachniss、Burgard，IEEE T-RO 2007，DOI `10.1109/TRO.2006.889486`；ROS Wiki 明确要求里程计并将每帧扫描变换到 `odom`，适合作为同输入主基线。
+- Karto 的 ROS 包 `ros-perception/slam_karto` 是 OpenKarto 的 ROS wrapper，`melodic-devel` 分支 package 版本为 0.8.1；ROS Wiki 明确有 `odom_frame`、扫描匹配和“偏离 odometry 的惩罚”参数。它与 SLAM Toolbox 同属 OpenKarto 系列，不应在主表中重复计数为两个完全独立的算法家族。
+- SLAM Toolbox 的论文为 Macenski、Jambrecic，JOSS 2021，DOI `10.21105/joss.02783`；官方 `noetic-devel` 分支存在，README 明确描述同时接收 laser scan 和 odometry，并用扫描匹配修正 odometry、构建 pose graph。它比旧 `slam_karto` 更适合作为 ROS1/Noetic 的现代 Karto 系主比较，但依赖较重。
+- RTAB-Map 的论文为 Labbé、Michaud，Journal of Field Robotics 2018，DOI `10.1002/rob.21831`；官方 `rtabmap_ros` 有 `noetic-devel` 分支，README 给出 Noetic 二进制安装方式并说明 lidar 使用。它可配置为仅 scan+odom 的 2D/ICP 模式，但默认生态更偏多传感器，适合作为可选外部参考，不宜直接与无相机/无 IMU 的 C3 混写成同条件主结果。
+- MRPT 的 `mrpt_icp_slam_2d`/`mrpt_rbpf_slam` 也有 ROS1 Noetic 包；官方 ROS Wiki 写明 ICP 2D 节点可使用水平激光、odom 可选，RBPF 版本接口类似 GMapping。由于依赖和参数体系较重，当前只列为备选，不列入第一轮主实验。
+- Hector-SLAM 论文为 Kohlbrecher 等，SSRR 2011，DOI `10.1109/SSRR.2011.6106777`；官方 ROS Wiki 明确写明“不需要里程计”，所以只能作为 LiDAR-only 辅助外部参考，不能替代 scan+odom 主基线。
+- 推荐主对比集合：`C0 clean Cartographer`、`GMapping`、`SLAM Toolbox`、`C3 proposed`。可选增加 `RTAB-Map(scan+odom, no camera/no IMU)`；`Hector-LiDAR-only` 单独标注输入条件。不要把 `laser_scan_matcher`/`rf2o` 当成完整 SLAM，也不要把 LIO-SAM、FAST-LIO、VINS 等需要 IMU/3D/相机的算法纳入同输入主表。
+- 本轮只完成联网检索和证据核对，没有安装、改代码或运行新算法。正式补跑前仍需先确认用户同意主对比集合，并锁定每个算法的参数、输入话题、外参和评价口径。

+- 源码审计发现最高优先级风险：创新点二将第 i 节点坐标系中的相邻位姿平移残差与 local-SLAM 坐标系中的退化方向直接点积，存在坐标系不一致；返程、掉头和转弯时风险最大。C2 与 C3 的 `motion_alignment_min_cosine` 也不一致，现有四组不是严格同构消融。修复这两项前不应继续冻结论文结果。
+- 创新点一的真实实现是“原始点云 PCA 退化代理 + 受 odom 影响的前端平移预测先验方向调权”，不是直接对 LiDAR 残差建立统计协方差。创新点二是“local-SLAM/odom 一致性代理 + 匹配分数门控 + 后端 odom 边分量降权”，且只在后端生效，坏 odom 仍进入前端外推和创新点一先验，存在反向耦合。
+- 权威文献显示，原始点云 PCA/特征值退化判断已有明显先例，不能单独作为强创新。关键参照包括：Censi 2007 ICP协方差、Zhang-Kaess-Singh 2016 Hessian退化子空间、X-ICP 2024逐方向可定位性、Probabilistic Degeneracy Detection 2024、GenZ-ICP 2025、LION 2021、2DLIW-SLAM 2024和Featureless Tunnel Wheel-Encoder SLAM 2023。
+- 轮速异常与统计一致性方面，关键参照包括 Innovation-based fault detection、Mahalanobis/联合兼容检验、CUSUM、Odometry 2.0；鲁棒后端方面包括 Switchable Constraints、Max-Mixtures、DCS、GNC和AEROS。它们多数不是同输入公平基线，但可用于重构当前方法的理论依据。
+- 当前最有价值且工作量可控的优化路线是：先修坐标系和严格消融；再用实际 scan-matcher Jacobian/Hessian 形成方向可观测性和 LiDAR 协方差；以 `NIS = r^T(Σ_wheel+Σ_lidar)^-1r` 替代米/弧度混合固定阈值，并用 CUSUM/迟滞识别持续异常；最后在 odom 边增加 DCS 类漏检保护，并把统一的 odom trust 反馈到前端，避免后端降权时前端仍增强坏 odom 先验。
+- 论文可守住的新颖性方向应收敛为：面向无 IMU 的 2D LiDAR+wheel odom，在统一的方向协方差/可靠性框架下协调 LiDAR 几何退化与轮速异常，避免一方退化时错误强化另一方；不能继续把“PCA检测退化”或“固定阈值降权”单独包装为主要创新。

+## 2026-08-04 A'+B' 升级方案漏洞复审
+
+- 最高风险1：不能从包含 occupied-space、translation prior 和 rotation prior 的完整 Ceres problem 直接取 Hessian。当前先验残差位于 `ceres_scan_matcher_2d.cc:514-531`，会人为补足弱方向，使退化环境看起来良态。可观测性只能由 LiDAR occupied-space 残差的 Jacobian/信息矩阵单独计算。
+- 最高风险2：标准 `NIS=r^T(Σ_lidar+Σ_odom)^-1r` 默认 LiDAR 与 odom 独立，但当前 odom 同时参与 PoseExtrapolator、scan matcher 初值和运动先验，因此两者明显相关；直接相加协方差会过度自信并误报。若不显式估计交叉协方差，就不能称严格 NIS/卡方检验，应改为在 LiDAR 可观子空间内、使用开发集经验协方差标定的 normalized consistency statistic。
+- 最高风险3：Hessian 的时间因果链未定义。优化前位姿不准，优化后 Hessian 又来不及控制同一次 solve；完整同帧控制需要两阶段匹配或自定义迭代，工作量和实时开销显著。推荐最小可行方案是优化后只评估 LiDAR-only 信息，结果延迟一帧用于下一帧前端和当前/后续后端边，并明确一帧延迟。
+- 最高风险4：当前后端每次优化都会重新创建 `SlipDetector` 并重放全部历史边，不能作为实时前端 trust 的状态来源。CUSUM/trust 管理器必须位于前端因果数据流，后端只消费按节点时间固化的 trust；否则后端批量重放会产生不可解释的在线反馈和重复状态。
+- 高风险5：只有 LiDAR 与聚合 `/odom` 两源时，不一致只能说明“跨源不一致”，不能在 LiDAR 也不可靠时识别为真实 wheel slip。论文和话题应改称 odometry inconsistency/trust，而非已确认 slip；Wheel-float02 bag也没有左右轮速或转矩可用于轮级诊断。
+- 高风险6：状态不能用全局四个二值状态粗略实现。`x/y/yaw` 的可观性和 odom trust必须分方向、连续化；近似各向同性时特征向量不稳定，还需处理量纲归一化、特征向量符号、方向连续性和坐标变换。
+- 高风险7：把 DCS 独立应用到连续 odom 边可能在一段持续异常时同时削弱大量相邻边，造成图局部欠约束。DCS只能后置、设置权重下限并单独消融，不能在首轮和 trust 逻辑同时上线。
+- 高风险8：前端降低 odom trust 后必须定义替代预测。Cartographer 当前有 `linear_velocity_from_poses_`，但只在 odom 数据不足时使用；推荐连续混合 `v_pred=α_o v_odom+(1-α_o)v_pose`，不能硬切换或直接冻结，否则会引入位姿跳变。
+- 实验风险：当前 RPE不是标准SE(2)相对位姿误差，Mocap到base的杠杆臂又未明确；评价修复必须先于算法。Wheel01只能开发、Wheel02锁定测试；阈值或经验协方差不得用Wheel02调。B-only依赖共同的LiDAR可观测性观测器，因此四组消融需把观测器设为所有组共有的只读基础设施，只消融“前端方向动作”和“trust动作”。
+- 复审后修正的最小路线：Phase 0先修评价、坐标系和同构消融；Phase 1仅实现并验证LiDAR-only归一化信息矩阵，不改权重；只有该指标能预测退化区误差时才进入Phase 2；Phase 2用经验归一化一致性统计和前端一帧延迟的分方向trust，连续混合odom/pose历史预测，并把固化trust传给后端；Phase 3只有确认仍有漏检时才单独加入DCS。
+- 原先“10--15个工作日完成完整方案”的估计偏乐观。严格实现、单测、诊断和最小实验约需15--25个有效工作日，完整冻结实验与论文重写另计。九月投稿时间下必须设置Phase 1/2停止门槛，不能同时开发P2d、主动特征和DCS。

+## 2026-08-04 物理可观测性与 A'+B' 方案复审结论
+
+- 严格结论是“需要调整，但不推翻 Phase 0/1”。当前 LiDAR-only occupied-space `J^T J` 诊断和后端坐标系修复仍然是必要的正确性基础；尚未实施的严格 NIS、CUSUM、trust 反馈和 DCS 不应按原描述直接上线。
+- 真正的物理极限不是“交叉协方差无法消除”，而是长走廊轴向 LiDAR 观测近似不变、且 wheel odom 同方向受持续偏置时，现有两源不再提供足以恢复真实轴向位移的独立信息。交叉协方差可以显式估计或保守处理，但无法凭空创造缺失的物理观测。
+- 当前源码确认 odom 会进入 `PoseExtrapolator`，而 scan-to-map 同时包含相关搜索、occupied-space 残差和运动先验。因此 LiDAR 结果与 odom 并不独立，但 LiDAR 也不是必然被 odom 带偏：存在端墙、门框、转角、局部地图结构或有效回环时，scan-to-map 仍可纠正外推初值。
+- 不显式建模交叉协方差时，不能将 `r^T(Σ_lidar+Σ_odom)^-1r` 称为统计严格、可按卡方分布标定的 NIS。Phase 2 应改为“LiDAR 可观子空间内的经验归一化跨源一致性统计”，只输出 odometry inconsistency/trust，不宣称已识别真实 wheel slip。
+- 一帧延迟不是“完全无效”：它无法防住异常首帧，对极短脉冲的相位跟踪也有限，但对持续多帧退化仍能减少后续错误累积。因此必须报告检测延迟、持续时间分层和首帧无保障的边界，不能宣称实时根治瞬态打滑。
+- 论文不应降格为只做“失败预警”，否则与当前 SLAM 性能优化的主题和实验目标不一致。可守住的主张应是“可观性感知的方向融合与失效约束”：在仍有信息的区间改善 APE/RPE 或至少非劣，在双源同向不可信时限制错误增长并诚实输出不确定性，不承诺恢复不可观真值。
+- `Failure Time` 只能作为预注册失败阈值下的辅助鲁棒性指标，不能替代 Wheel-float01/02 的 APE/RPE、尾部误差、失败率和非劣性证据。Corridor02 继续只用结构指标，`corridor_new.bag` 继续排除。
+- Phase 2 新的入口门槛：先在 Wheel-float01 验证 LiDAR-only 指标与局部真值误差/已知几何退化有时序关联，且能区分“LiDAR 可观时的跨源冲突”和“LiDAR 不可观时的未知冲突”。若做不到，不实施 trust 控制，转向结构特征增强路线。
+- 本次仅完成只读源码/实验记录审查与方案调整，未修改任何算法代码、参数或 launch，未启动新实验。

+## 2026-08-04 调整后方案的第二轮严格审核
+
+- 新发现的首要漏洞：当前所谓 LiDAR-only `J^T J` 虽然只选择 occupied-space 残差行，但是在包含 translation/rotation prior 的联合 Ceres solve 之后、其优化位姿处评估；参考 probability-grid submap 也由受 odom 影响的历史融合位姿构建。因此它是“融合位姿处的 LiDAR 残差局部曲率”，不是与 odom 统计独立的 LiDAR 位姿观测。
+- 第二个漏洞：当前诊断只对 `H` 的 2x2 平移块做特征分解，并单独记录 `H(2,2)` 作为 yaw 信息，忽略了平移-yaw 耦合项。长走廊中可能出现单独 2x2 块看似良态、但完整 SE(2) 可观性仍较弱的情况。后续必须使用无量纲化的 3x3 信息或经 yaw 耦合消元后的平移 Schur complement，不能直接将当前特征值称为位姿协方差。
+- `J^T J` 还受 occupied-space weight、栅格分辨率/插值、submap 成熟度和点云筛选影响；`1/sqrt(N)` 只控制点数尺度，不能将 Gauss-Newton 曲率自动变成标定协方差。该量必须用 Wheel-float01 经验标定并做配置敏感性审查。
+- 前端 trust 反馈还有系统集成漏洞：Cartographer 同时有 local trajectory builder 内部的扫描匹配外推器，以及 ROS Node 中用于 TF 发布的外推器；同一份 odom 同时进入两者。若 trust 只修改一处，映射位姿和对外 TF 可能不一致。实施前必须明确 trust 在两条路径中的作用边界和时间戳。
+- 用历史 pose 速度替代 odom 不是新传感器信息；该 pose 同样受历史 odom 和 submap 影响。连续混合只能用于减少跳变和限制错误传播，不能包装为恢复缺失位移。
+- 经验 trust 阈值在 baseline 地图上标定后，开启反馈会改变轨迹和 submap，进而改变诊断量分布；这是一个闭环分布偏移问题。必须冻结 Wheel-float01 参数，在 Wheel-float02 只测一次主结果，并补充阈值敏感性，不能在测试集闭环调参。
+- 完成性判断必须分开：“代码完成”在不加 DCS、P2d 或主动特征的前提下仍然可行；“完成并证明明显收益”不能预先保证。截止 2026-08-04，九月前约一个月，只能执行一条最小路线：先 1--2 个有效工作日完成 Wheel-float01 诊断门槛，通过后再实施经验一致性和最小 trust；DCS/CUSUM/结构特征不应同时开发。
+- 创新性结论是“条件性、中等强度的硕士级系统组合创新”：只有当论文展示统一的方向可观性状态、LiDAR 可观门控的 odom 一致性、因果固化的前后端 trust，并通过严格消融证明协同收益，才能成立。若最终只是 Hessian 阈值 + odom 降权 + DCS，则新颖性不足。
+- 预期收益上，trust 方案主要是“避免错误强化和限制退化”，它不会创造新几何特征。因此在所有指标上肉眼明显、大幅超过 Hector/其他算法的目标上，当前方案成功概率不高；它更合理的目标是退化片段的显著尾部误差降低、全程 APE 非劣和失败率下降。
+- 本轮仅完成只读审查并更新本日总结，未修改任何算法代码、参数或 launch，未启动新实验。

+## 2026-08-04 OAD-CSM 候选主线漏洞复审
+
+- 最致命的漏洞是参考地图污染：`LocalTrajectoryBuilder2D::ScanMatch()` 只匹配 `active_submaps_.submaps().front()`，而当前 scan 又会按该匹配位姿继续插入所有活动 submap。若长时间退化中 odom 持续偏置，submap 也会被拉伸/扭曲；此时一个清晰的候选单峰可能只是“错误轨迹与错误地图的自洽”，不代表重捕获真值。
+- 活动 submap 由 `num_range_data=70` 控制：新 submap 在最新 submap 收满 70 次插入时创建，老 submap 在 140 次插入时完成。因此 OAD-CSM 只可能在参考结构仍保持足够重叠、且 submap 尚未严重变形的有限时间/距离内重捕获，不能泛化为长时间走廊漂移解决方案。
+- 候选地图分数是占用概率均值，不是归一化概率似然。峰宽、熵或候选协方差还受栅格分辨率、窗口截断、角度采样、地图成熟度和动态物体影响，只能称为经验歧义指标，不能直接当作统计协方差。
+- 直接用 top-1/top-2 差值也有漏洞：网格上相邻候选高度相关，次高候选通常只是主峰邻居。必须先非极大值抑制/峰聚类，再计算主峰与独立次峰的差距；多峰分布不能被一个协方差概括。
+- 自适应窗口的因果链仍未完整：本帧尚未扩窗前不知道完整分数面的弱方向；用上一帧方向又受场景转换延迟影响。可行实现需要“小窗探测 -> 边界/宽峰检查 -> 有界粗到细扩窗”，而非单次扩窗。
+- 实时性是硬约束：现有 0.15 m 窗口在 0.05 m 栅格下每个角度层约 49 个平移候选；扩大到 1 m 各向同性窗口会增至约 1681 个。在当前±20°角度窗下候选数会大幅上升。必须使用方向带状过滤、动态缩小 yaw 窗口或粗到细搜索，并把延迟/丢帧列入主实验。
+- 上一版的“暂缓 submap 插入”并不是小保护：跳过插入会导致地图缺口、节点/约束减少和 submap 生命周期停滞；缓存并回放扫描则需要重构前端数据流。该功能应从九月前最小方案中删除。
+- 创新性也被上述限制削弱：分数面歧义、自适应搜索窗和峰值门控各自属于已知技术思路。只有在无 IMU 2D LiDAR+wheel odom 中形成“方向带状重捕获 + 地图/先验分数解耦 + 实时有界扩窗”并证明相对固定窗和单纯扩窗的协同收益，才可能作为中等强度组合创新。
+- 复审后的严格定位：OAD-CSM 是一个“有界局部重捕获增强”，比 trust-only 更可能产生局部收益，但不能独立解决长时间 submap 共同漂移。当前不建议直接将它冻结为论文主线。
+- 新的先决门槛：诊断必须在 Wheel-float01 记录候选的纯地图分数、窗口边界、活动 submap 年龄/插入计数，并将候选位姿与 Mocap 真值对照。只有当真值附近的候选在多个非重叠事件中获得稳定独立高峰，且该高峰不是已污染 submap 的偏差自洽，才可进入控制实现。
+- 本轮只完成源码与方案复审，未修改算法、参数或 launch，未启动新实验。

+void UpdateLidarOnlyObservabilityMetric(
+    ceres::Problem* const problem,
+    const ceres::ResidualBlockId occupied_space_residual_block,
+    DirectionalDegeneracyMetric* const metric) {
+  metric->lidar_only_observability_enabled = true;
+  if (occupied_space_residual_block == nullptr) {
+    return;
+  }
+
+  ceres::Problem::EvaluateOptions evaluate_options;
+  evaluate_options.residual_blocks.push_back(occupied_space_residual_block);
+  evaluate_options.apply_loss_function = false;
+  ceres::CRSMatrix jacobian;
+  if (!problem->Evaluate(evaluate_options, nullptr, nullptr, nullptr,
+                         &jacobian) ||
+      jacobian.num_cols != 3 || jacobian.num_rows <= 0 ||
+      jacobian.rows.size() != static_cast<size_t>(jacobian.num_rows + 1)) {
+    return;
+  }
+
+  Eigen::Matrix3d information = Eigen::Matrix3d::Zero();
+  for (int row = 0; row < jacobian.num_rows; ++row) {
+    Eigen::Vector3d jacobian_row = Eigen::Vector3d::Zero();
+    for (int index = jacobian.rows[row]; index < jacobian.rows[row + 1];
+         ++index) {
+      const int column = jacobian.cols[index];
+      if (column >= 0 && column < 3 && std::isfinite(jacobian.values[index])) {
+        jacobian_row[column] = jacobian.values[index];
+      }
+    }
+    information.noalias() += jacobian_row * jacobian_row.transpose();
+  }
+
+  const Eigen::Matrix2d translation_information =
+      0.5 * (information.topLeftCorner<2, 2>() +
+             information.topLeftCorner<2, 2>().transpose());
+  Eigen::SelfAdjointEigenSolver<Eigen::Matrix2d> solver(
+      translation_information);
+  if (solver.info() != Eigen::Success || !solver.eigenvalues().allFinite() ||
+      !solver.eigenvectors().allFinite()) {
+    return;
+  }
+
+  const double minimum_information = std::max(0., solver.eigenvalues()(0));
+  const double maximum_information = std::max(0., solver.eigenvalues()(1));
+  constexpr double kInformationEpsilon = 1e-12;
+  if (maximum_information <= kInformationEpsilon) {
+    return;
+  }
+  metric->lidar_only_observability_valid = true;
+  metric->lidar_only_translation_min_information = minimum_information;
+  metric->lidar_only_translation_max_information = maximum_information;
+  metric->lidar_only_translation_condition_number =
+      maximum_information / std::max(kInformationEpsilon, minimum_information);
+  metric->lidar_only_yaw_information = std::max(0., information(2, 2));
+  metric->lidar_only_weak_translation_direction =
+      solver.eigenvectors().col(0).normalized();
+}

+- 推荐优化边界：只针对创新点二增加或强化“长走廊强退化时保护 odom 权重”的门控规则；原则是在 LiDAR 纵向不可观测时，不允许算法过度降低 odom 约束，避免把退化误当成打滑。
+- 验证边界：优化后必须在 `corridor_new.bag` 上做 clean baseline、innovation1-only、full proposed 对比；同时复验 `corridor_repeat_01/02/03` 和 M3DGR 既有冻结数据，确保没有反向影响。
+- 论文决策：如果小优化有效，可作为最终冻结版；如果无效，应停止继续调参，把 `corridor_new.bag` 写成局限性，主论文证据仍围绕已冻结的结构改善和异常鲁棒性实验。
+
+## 2026-07-31 是否继续优化的最终决策校准
+
+- 进一步校准后的结论是：不是要“继续把系统做得更复杂”，而是做一次受控修正，验证两个创新点联用时是否会在强退化长走廊里反向影响结果。
+- 若这次受控修正不能改善 `corridor_new.bag`，就应停止在该方向继续投入时间，把主线收束到已冻结证据。
+
+## 2026-07-31 其他 bag 的处理策略
+
+- 其他 bag 不能按“每个 bag 单独调一套参数”处理，否则会破坏论文口径一致性。
+- `corridor_repeat_01/02/03.bag` 应作为主回归测试集：要求任何新优化不能让结构代理指标回退，也不能引入 FATAL、IMU 订阅或不稳定发布。
+- `Corridor02` 应作为跨数据集泛化测试：只看结构代理指标，不把它写成连续真值精度证据。
+- `corridor_new.bag` 应作为挑战/失败样例：重点判断新修正是否能避免多走廊、重影和反向影响。
+- 若后续使用 `Wheel-float02`，它应承担唯一强真值验证角色；其他 bag 只作辅助和泛化支撑，不能混用成同一种证据。
+
+## 2026-07-31 总体是否继续优化的最终判断
+
+- 综合自有 repeat 三包、Corridor02 结构指标和 `corridor_new.bag` 失败样例后，最终判断是：值得继续做一轮受控优化，但不值得继续做大规模扩展优化。
+- 继续优化的唯一理由是修正真实非对称长走廊中的反向影响风险；如果这一轮修正不能改善 `corridor_new.bag`，就应冻结当前版本并把该 bag 作为局限性证据。
+- 不能继续优化的部分是：不能再为追更好数值而大范围调参，也不能把每个 bag 都调成单独最优，否则论文口径会失真。
+
+## 2026-08-03 消融实验、外部算法和 IMU 参考组设计建议
+
+- 消融实验推荐四组，而不是三组：`C0 baseline`、`C1 baseline+A`、`C2 baseline+B`、`C3 baseline+A+B`。四组才能分别说明创新点一、创新点二和联合运行收益。
+- 由于当前创新点二使用创新点一提供的 LiDAR 可靠性门控，`baseline+B` 必须明确实现为“B 使用独立/固定可靠性门控的单独版本”；不能把依赖 A 的 B 直接伪装成完全独立消融。

TEST(OptimizationProblem2DFrameTest, IdentityKeepsLocalDirection) {
  const Eigen::Vector2d direction(2., -1.);
  const Eigen::Vector2d converted = TransformLocalSlamDirectionToFirstNodeFrame(
      transform::Rigid2d::Identity(), direction);
  EXPECT_NEAR(converted.x(), direction.normalized().x(), 1e-12);
  EXPECT_NEAR(converted.y(), direction.normalized().y(), 1e-12);
}

TEST(OptimizationProblem2DFrameTest, InverseFirstNodeRotationIsApplied) {
  const double half_pi = std::acos(-1.) * 0.5;
  const Eigen::Vector2d direction_in_local_slam(0., 1.);
  const Eigen::Vector2d converted = TransformLocalSlamDirectionToFirstNodeFrame(
      transform::Rigid2d::Rotation(half_pi), direction_in_local_slam);

  // A +90 degree first-node pose maps tracking X to local Y. Therefore local
  // Y is tracking X when the relative edge is evaluated at the first node.
  EXPECT_NEAR(converted.x(), 1., 1e-12);
  EXPECT_NEAR(converted.y(), 0., 1e-12);
}

TEST(OptimizationProblem2DFrameTest, InvalidDirectionReturnsZero) {
  const Eigen::Vector2d converted = TransformLocalSlamDirectionToFirstNodeFrame(
      transform::Rigid2d::Identity(), Eigen::Vector2d::Zero());
  EXPECT_DOUBLE_EQ(converted.x(), 0.);
  EXPECT_DOUBLE_EQ(converted.y(), 0.);
}
